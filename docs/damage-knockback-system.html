1. Обзор системы

При контакте волны оружия с врагом: наносится урон, отображается HP бар, враг отбрасывается назад.

Код — hp_bar.gd

extends ProgressBar

@onready var health_component : HealthComponent = "../HealthComponent"

func _ready() -> void:
    max_value = health_component.max_health
    value = health_component.health
    health_component.health_changed.connect(_on_health_changed)

func _on_health_changed(new_health : float) -> void:
    value = new_health

    # Цвет зависит от процента HP
    var ratio = new_health / max_value
    if ratio > 0.5:
        modulate = Color.GREEN
    elif ratio > 0.25:
        modulate = Color.YELLOW
    else:
        modulate = Color.RED

Принцип работы

Волна оружия при контакте с врагом передаёт ему скорость через velocity. Враг — CharacterBody2D, поэтому move_and_slide() обрабатывает физику.

Волна летит
→
body_entered
→
направление = волна→враг
→
enemy.velocity += dir * force

Код — weapon_wave_controller.gd (обновлённый)

class_name WeaponWaveController
extends StaticBody2D

@export var move_speed : float = 600
@export var destroy_time : float = 1
@export var damage : float = 30 # ← NEW: урон волны
@export var knockback_force : float = 200 # ← NEW: сила отбрасывания
var can_move : bool = false
var direction : Vector2

func _ready() -> void:
    get_tree().create_timer(destroy_time).timeout.connect(queue_free)
    body_entered.connect(_on_body_entered) # ← NEW

func set_direction(dir : Vector2):
    direction = dir.normalized()
    rotation = direction.angle()
    can_move = true
    $AnimationPlayer.play("RESET")

func _on_body_entered(body : Node2D) -> void:
    if body is Enemy:
        # Нанести урон
        body.health_component.take_damage(damage)
        
        # Отбросить
        var knockback_dir = (body.global_position - global_position).normalized()
        body.velocity += knockback_dir * knockback_force

func _process(delta: float) -> void:
    if not can_move:
        return
    global_position += direction * move_speed * delta

5. Числа урона NEW

Опционально: при попадании показывать числа урона, которые всплывают и исчезают.

Код — damage_number.gd

extends Label

var velocity : Vector2 = Vector2(0, -80)
var lifetime : float = 0.8
var elapsed : float = 0

func _ready(amount : float) -> void:
    text = "-" + str(int(amount))
    modulate = Color(1, 0.3, 0.3) # красный

func _process(delta : float) -> void:
    elapsed += delta
    position += velocity * delta
    velocity.y += 200 * delta # гравитация
    modulate.a = 1 - (elapsed / lifetime)
    
    if elapsed >= lifetime:
        queue_free()
